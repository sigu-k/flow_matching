import torch
import numpy as np

def sample_t(n, dist, eps=0.1):
    if dist == 'uniform':
        return torch.rand(n)
    params = {
        'center': (0.0, 1.2),
        'both':   (0.0, 2.0),
        'data':   (-0.5, 1.5),
        'noise':  (0.5, 1.5),
    }
    mu, sigma = params[dist]
    z = torch.randn(n)
    t_ln = torch.sigmoid(mu + sigma * z)
    t_uni = torch.rand(n)
    mask = (torch.rand(n) < eps)
    return torch.where(mask, t_uni, t_ln)

torch.manual_seed(0)
n = 200000
print('最終確定版5パターン(eps=0.1, n=200k)')
print('='*70)
for dist in ['uniform','center','both','data','noise']:
    t = sample_t(n, dist).numpy()
    hist,_ = np.histogram(t, bins=10, range=(0,1))
    bar = ' '.join(f'{h//2000:2d}' for h in hist)
    mn = (t<0.05).sum()/n*100
    mx = (t>0.95).sum()/n*100
    allpos = (hist>0).all()
    print(f'{dist:8s}: median={np.median(t):.2f} 端[<0.05:{mn:4.1f}% >0.95:{mx:4.1f}%] 全区間>0:{allpos}')
    print(f'          [{bar}]')
